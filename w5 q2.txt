#include <iostream>
#include <cstdlib>
#include <ctime>
#include <vector>
#include <stack>

#define SIZE 10

using namespace std;

class Node {
public:
    Node() : row(0), col(0) {}
    Node(int r, int c) : row(r), col(c) {}

    int getRow() { return row; }
    int getCol() { return col; }
    void setRow(int r) { if (r >= 0 && r < SIZE) row = r; }
    void setCol(int c) { if (c >= 0 && c < SIZE) col = c; }

private:
    int row, col;
};

class List {
public:
    List() : top(0) {}

    void addElement(int r, int c) {
        if (top < SIZE * SIZE) {
            data[top++] = Node(r, c);
        }
    }

    Node* removeElement() {
        if (top > 0) {
            return new Node(data[--top]); // return a new pointer to the removed element
        }
        return nullptr; // return NULL if list is empty
    }

    void printList() {
        for (int j = 0; j < top; j++) {
            cout << "(" << data[j].getRow() << ", " << data[j].getCol() << ")" << endl;
        }
    }

private:
    Node data[SIZE * SIZE];
    int top;
};

class Maze {
public:
    Maze() {
        initMaze(SIZE);
    }

    void initMaze(int s) {
        maze = new int*[s];
        for (int i = 0; i < s; i++) {
            maze[i] = new int[s];
        }

        // Initialize all cells to 0
        for (int i = 0; i < s; i++) {
            for (int j = 0; j < s; j++) {
                maze[i][j] = 0;
            }
        }

        // Place walls randomly (20%)
        int wallCount = (s * s) / 5; // 20% of the cells as walls
        srand(time(0));
        for (int i = 0; i < wallCount; i++) {
            int r = rand() % s;
            int c = rand() % s;

            // Make sure [0][0] and [s-1][s-1] are empty
            if ((r == 0 && c == 0) || (r == s - 1 && c == s - 1)) continue;
            maze[r][c] = 1; // Set wall
        }
    }

    List* getPath() {
        List* path = new List();
        bool visited[SIZE][SIZE] = { false };
        stack<Node> stack;

        // Start from (0, 0)
        stack.push(Node(0, 0));
        visited[0][0] = true;

        // Directions for moving in the maze (right, down, left, up)
        int directions[4][2] = { {0, 1}, {1, 0}, {0, -1}, {-1, 0} };

        while (!stack.empty()) {
            Node current = stack.top();
            path->addElement(current.getRow(), current.getCol());
            stack.pop();

            // If we've reached the end
            if (current.getRow() == SIZE - 1 && current.getCol() == SIZE - 1) {
                return path; // Return the path found
            }

            // Explore neighbors
            for (auto& dir : directions) {
                int newRow = current.getRow() + dir[0];
                int newCol = current.getCol() + dir[1];

                if (newRow >= 0 && newRow < SIZE && newCol >= 0 && newCol < SIZE &&
                    maze[newRow][newCol] == 0 && !visited[newRow][newCol]) {
                    visited[newRow][newCol] = true;
                    stack.push(Node(newRow, newCol));
                }
            }
        }

        // If no path found, return empty list
        delete path;
        return new List(); // Return empty list
    }

    void printMaze() {
        for (int j = 0; j < SIZE; j++) {
            for (int k = 0; k < SIZE; k++) {
                if (maze[j][k] == 0)
                    cout << " ";
                else
                    cout << "*";
            }
            cout << "\n";
        }
    }

    ~Maze() {
        for (int i = 0; i < SIZE; i++) {
            delete[] maze[i];
        }
        delete[] maze;
    }

private:
    int** maze;
};

int main() {
    Maze* maze = new Maze();
    maze->printMaze();
    maze->getPath()->printList();

    delete maze; // Clean up the maze
    return 0;
}
