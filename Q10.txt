#include <iostream>
#include <cmath>
#include <vector>

using namespace std;

template<class T>
class TreeInLinkedList {
private:
    class TreeNode {
    public:
        TreeNode(T d, TreeNode *p) : data(d), parent(p) {}
        TreeNode *parent;
        T data;
    };

    vector<TreeNode *> *nodeList;

    void displayPreorder(TreeNode *node) {
        if (node) {
            cout << node->data << " ";
            // Display all children of the current node
            for (TreeNode *child : *nodeList) {
                if (child->parent == node) {
                    displayPreorder(child);
                }
            }
        }
    }

    void displayPostorder(TreeNode *node) {
        if (node) {
            // Display all children of the current node first
            for (TreeNode *child : *nodeList) {
                if (child->parent == node) {
                    displayPostorder(child);
                }
            }
            cout << node->data << " ";  // Display the node after its children
        }
    }

public:
    TreeInLinkedList() {
        nodeList = new vector<TreeNode *>();
    }

    ~TreeInLinkedList() {
        for (TreeNode *node : *nodeList) {
            delete node;
        }
        delete nodeList;
    }

    void addElement(T data) {
        int j, k = nodeList->size();
        if (data == 1) {
            for (TreeNode *node : *nodeList) {
                delete node;  // Cleanup existing nodes
            }
            nodeList->clear();
            TreeNode *newNode = new TreeNode(data, nullptr);
            nodeList->push_back(newNode);
        } else {
            for (j = 0; j < k; j++) {
                if (data % (*nodeList)[j]->data == 0) {
                    TreeNode *newNode = new TreeNode(data, (*nodeList)[j]);
                    nodeList->push_back(newNode);
                }
            }
        }
    }

    void displayPreorder() {
        if (!nodeList->empty()) {
            displayPreorder((*nodeList)[0]); // Start from the root
        }
    }

    void displayPostorder() {
        if (!nodeList->empty()) {
            displayPostorder((*nodeList)[0]); // Start from the root
        }
    }
};

int main() {
    TreeInLinkedList<int> *tree = new TreeInLinkedList<int>();
    int j;
    int n;
    cin >> n;
    for (j = 1; j <= n; j++) {
        tree->addElement(j);
    }
    tree->displayPreorder();
    cout << endl;
    tree->displayPostorder();
    cout << endl;

    delete tree; // Clean up the tree
    return 0;
}
