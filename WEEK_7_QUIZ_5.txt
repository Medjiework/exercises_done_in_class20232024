#include <stdio.h>

int skyline[10001], l, t, r, i, f;

int main() { //loop until no rectangle
  while (scanf("%d %d %d" , &l, &t, &r) == 3) {
    //update the skyline
    for (i = l; i < r; i++) {
      if (skyline[i] < t) {
        skyline[i] = t;
      }
    }
  }
  //print skyline after all rectangle
  for (i = 0, f = 1; i <= 10001; i++) {
    if (skyline[i - 1] != skyline[i]) {
      if (f == 1) {
        printf("%d %d", i, skyline[i]);
        f++;
      } else {
        printf(" %d %d", i, skyline[i]);
      }
    }
  }
  //to separate diff case
  printf("\n");
  
  return 0;
}
